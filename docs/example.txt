/*
 * Sample Rill Script
 * Demonstrates the syntax and features of the Rill language.
 *
 * This example models a DTN bundle processing pipeline, but Rill is a
 * general-purpose embedded scripting language — not DTN-specific.
 */

// Extern namespace dependencies (provided by the embedder)
require bpsec;                                  // BPSec validation functions
require cbor;                                   // CBOR encode/decode
require time;                                   // Time functions
require codes;                                  // Bundle status report reason codes

// Source file imports (local .rill files)
import "../common/validation.rill";             // Namespace: validation
import "./helpers/routing.rill" as routing;     // Namespace: routing (explicit alias)

// File-scope variables (globals). A never-written global with a foldable
// initializer is the replacement for `const` — the optimizer inlines it to a
// constant. Accessed inside functions via `::NAME`. No destructuring at file
// scope, so flags are declared individually.
let MAX_AGE = 86400;                            // 24 hours in seconds
let DEFAULT_PRIORITY = 5;
let URGENT_PRIORITY = DEFAULT_PRIORITY * 2;     // Initializers may use earlier globals
let MAX_HOPS = 16;
let FLAG_URGENT = 0x01;
let FLAG_PROCESSED = 0x02;
let FLAG_SECURITY = 0x04;

// ============================================================================
// Entry Points (callable by the embedder via FunctionHandle)
// ============================================================================

/*
 * Checks bundle age and drops if too old.
 * Demonstrates: extern calls, constants, if expressions, exit.
 */
fn check_age(bundle) {
    // Use imported function from source module
    if !validation::is_valid_timestamp(bundle.creation_time) {
        exit(codes::BLOCK_UNINTELLIGIBLE);
    }

    // Use extern function and local constant
    let current = time::now();
    if current - bundle.creation_time > ::MAX_AGE {
        exit(codes::LIFETIME_EXPIRED);
    }
}

/*
 * Validates BPSec signatures.
 * Demonstrates: for loops, extern namespace calls.
 */
fn validate_security(bundle) {
    if let blocks = bundle.security_blocks {
        for block in blocks {
            if !bpsec::validate_signature(block, bundle) {
                exit(codes::FAILED_SECURITY_OPERATION);
            }
        }
    }
}

/*
 * Processes a bundle: validates, normalizes, routes.
 * Demonstrates: with (reference bindings), if let, if with, type patterns.
 */
fn process(bundle) {
    // Validate required fields (helper can exit directly)
    validate_required_fields(bundle);

    // Validate CBOR well-formedness using extern
    if !cbor::is_well_formed(bundle.payload) {
        exit(codes::BLOCK_UNINTELLIGIBLE);
    }

    // Reference binding with `with`: mutations flow back to bundle
    with dest = bundle.destination;
    if dest.node == "mars" {
        bundle.route = routing::calculate(bundle);
    }

    // Type-narrowing reference: if bundle.priority is a UInt, mutate in place
    if with UInt(priority) = bundle.priority {
        priority += 1;
    }

    // Value binding with `if let`: converted is a copy, bundle unchanged
    if let converted = to_uint(bundle.metadata.priority) {
        if converted > 0 {
            bundle.priority = converted;
        }
    } else {
        bundle.priority = ::DEFAULT_PRIORITY;
    }

    // Chained conditions with && (all must succeed)
    if with source = bundle.source && with dest = bundle.destination {
        validation::check_route(source, dest);
    }

    // Normalize metadata (implicit return — last expression without ;)
    normalize_metadata(bundle)
}

// ============================================================================
// Pattern Matching
// ============================================================================

/*
 * Validates payload structure using match expressions.
 * Demonstrates: match, type patterns, array patterns, map patterns, guards.
 */
fn validate_payload(payload) {
    match payload {
        // Map pattern: match specific keys
        {type: Text("data"), content: content} => {
            if !is_valid_content(content) {
                exit(codes::BLOCK_UNINTELLIGIBLE);
            }
        }

        // Array pattern with rest: first element + remaining
        let [Text(header), ..rest] => {
            // `let` in match arm = by-value bindings
            if header != "VALID" {
                exit(codes::BLOCK_UNINTELLIGIBLE);
            }
            // rest is a Sequence (zero-copy view of remaining elements)
            let items = collect(rest);
            len(items)
        }

        // Type pattern with guard
        Text(s) if len(s) > 0 => {
            cbor::validate_text_encoding(s);
            len(s)
        }

        // Wildcard catch-all
        _ => {
            exit(codes::BLOCK_UNINTELLIGIBLE);
        }
    }
}

/*
 * Validates required fields using standard prelude functions.
 * Demonstrates: is_defined, is_uint, with + type patterns.
 */
fn validate_required_fields(bundle) {
    if !is_defined(bundle.destination) || !is_defined(bundle.source) {
        exit(codes::BLOCK_UNINTELLIGIBLE);
    }

    // Type pattern checks both existence and type in one step
    if with UInt(ttl) = bundle.ttl {
        if ttl == 0 {
            exit(codes::LIFETIME_EXPIRED);
        }
    }
}

/*
 * Content validation helper.
 * Demonstrates: match as expression (implicit return), len intrinsic.
 */
fn is_valid_content(content) {
    match content {
        Text(s) => {
            let length = len(s);
            length > 0 && length < 1024
        }
        UInt(n) => n > 0,
        _ => false
    }
}

/*
 * Normalizes metadata into a consistent map format.
 * Demonstrates: type checking, if let, map literals.
 */
fn normalize_metadata(bundle) {
    let raw = bundle.metadata;

    if is_map(raw) {
        return raw;
    }

    if let uint_val = to_uint(raw) {
        return {"value": uint_val};
    }

    // default() from standard prelude: returns value if defined, else fallback
    let debug_str = default(to_text(raw), "unknown");
    {"debug": debug_str, "type": "unknown"}
}

/*
 * Demonstrates loop types: for with ranges, arrays, maps, while, loop.
 */
fn process_routes(bundle) {
    let scores = [];

    // Range-based for loop (lazy sequence, O(1) memory)
    for i in 0..5 {
        scores = scores;  // placeholder
    }

    // For loop over array with by-reference binding (default)
    for hop in bundle.previous_hops {
        if is_blocked(hop) {
            continue;
        }
    }

    // For loop with by-value binding (let = copy)
    for let hop in bundle.previous_hops {
        // hop is a copy; mutations don't affect bundle.previous_hops
        hop += 1;
    }

    // For loop with key-value pairs over a map
    for key, value in bundle.metadata {
        if key == "priority" && is_uint(value) {
            return value
        }
    }

    // Inclusive range
    for i in 0..=10 {
        // yields 0, 1, 2, ..., 10
    }

    // While loop
    let index = 0;
    while index < len(bundle.options) {
        if is_valid_path(bundle.options[index]) {
            return bundle.options[index];
        }
        index += 1;
    }

    // Infinite loop with break (loop is an expression)
    let result = loop {
        let candidate = generate_path();
        if is_valid_path(candidate) {
            break candidate;   // break with value
        }
    };

    result
}

/*
 * Block expressions and implicit return.
 * Demonstrates: block as expression, if as expression, no trailing semicolon.
 */
fn compute_priority(bundle) {
    // Block as expression: last expression (no ;) is the value
    let base = {
        let urgency = bundle.flags & ::FLAG_URGENT;
        let importance = default(bundle.importance, 1);
        urgency * 10 + importance
    };

    // If as expression — also the function's return value (no ; or return)
    if bundle.age > 3600 {
        base * 2
    } else {
        base
    }
}

/*
 * Operators: arithmetic, comparison, logical, bitwise, bit test.
 */
fn calculate_score(data) {
    // Arithmetic
    let sum = data.a + data.b;
    let product = data.c * data.d;

    // Comparison (strict boolean — no truthiness)
    let comparison = sum > product;

    // Unary
    let negated = -sum;
    let inverted = !comparison;
    let bitwise_not = ~data.flags;

    // Bitwise
    let masked = data.flags & 0xFF;
    let combined = data.flags | ::FLAG_PROCESSED;
    let shifted = data.value << 2;

    // Bit test operator (@)
    let is_urgent = data.flags @ 0;         // true if bit 0 is set
    let is_secure = data.flags @ 2;         // true if bit 2 is set

    // Bit set (@ as lvalue)
    with flags = data.flags;
    flags @ 1 = true;                       // set bit 1
    flags @ 3 = false;                      // clear bit 3

    // Type cast (as) — infallible numeric reinterpretation
    let signed = data.counter as Int;       // UInt → Int bit reinterpret
    let precise = data.counter as Float;    // UInt → Float widen

    // Return with if expression
    if comparison { sum } else { product }
}

/*
 * Demonstrates: let/with destructuring, rest patterns, for with pairs.
 */
fn process_array(data) {
    // Array destructuring (by value)
    let [first, second, third] = data;

    // Array destructuring with rest (by value)
    let [head, ..tail] = data;
    // tail is a Sequence — zero-copy lazy view of remaining elements
    // collect() materializes it into an Array
    let tail_arr = collect(tail);

    // Rest in middle position
    let [start, ..middle, end] = data;

    // Reference destructuring (by reference — mutations flow back)
    with [a, b] = data;
    a += 1;     // mutates data[0]
    b += 1;     // mutates data[1]

    // For loop with index-element pairs
    for i, element in data {
        // i is always by-value (index), element is by-reference (default)
        if i == 0 {
            element = 999;  // mutates data[0]
        }
    }

    len(data)
}
